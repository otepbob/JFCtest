Setup Guide & Integration Instructions

☁️ 3-Minute Microsoft OneDrive Setup Guide (Power Automate):

  1. In your OneDrive for Business / SharePoint, create an Excel file named Jabra_Floorwalk_Logs.xlsx.
  2. Inside the file, create a Table (select cells and press Ctrl+T) named FloorwalkTable with columns:
    Record_ID, Inspection_Date, Inspection_Time, Inspector, Building, Floor, Huddle_Room, Status, Remarks, Checked_Time, Submitted_At
  3. Go to make.powerautomate.com → Create → Instant cloud flow → Choose When an HTTP request is received.
  4. In the trigger, paste the JSON payload schema below (click "Copy Payload Schema").
  5. Add an action: Excel Online (Business) > Add a row into a table:
    - File: Select your Jabra_Floorwalk_Logs.xlsx in OneDrive
    - Table: Select FloorwalkTable
    - Map the row parameters to the trigger's details array.
  6. Save the Flow, copy the generated HTTP POST URL, and paste it into the URL field above!

} // Register this alert as PENDING in Pending_Alerts. Any earlier PENDING alert for the same // record OR the same site (a newer submission within the delay window) is marked SUPERSEDED, // so only the latest alert is sent. var alertId = Utilities.getUuid(); var site = params.site || ""; var lock = LockService.getScriptLock(); lock.waitLock(30000); var alertSheet = ss.getSheetByName("Pending_Alerts"); if (!alertSheet) { alertSheet = ss.insertSheet("Pending_Alerts"); alertSheet.appendRow(["Record_ID", "Inspector", "To_Email", "Date", "Remarked_Rooms", "Sent_At", "Status", "Alert_ID", "Site"]); alertSheet.getRange("A1:I1").setFontWeight("bold").setBackground("#e2e8f0"); } var existingRows = alertSheet.getDataRange().getValues(); for (var er = 1; er < existingRows.length; er++) { if (existingRows[er][6] === "PENDING" && (existingRows[er][0] === params.recordId || (site && String(existingRows[er][8]) === String(site)))) { alertSheet.getRange(er + 1, 7).setValue("SUPERSEDED"); } } alertSheet.appendRow([params.recordId, inspector, toEmail, date, params.remarkedRooms || "[]", "", "PENDING", alertId, site]); SpreadsheetApp.flush(); lock.releaseLock(); // Sleep for the delay duration, then check if THIS alert is still PENDING Utilities.sleep(delayMinutes * 60 * 1000); // Verify 1: Our own row must still be PENDING (not CANCELLED / SUPERSEDED) var ownRow = -1; var aRows = alertSheet.getDataRange().getValues(); for (var ar = aRows.length - 1; ar >= 1; ar--) { if (aRows[ar][7] === alertId) { ownRow = ar + 1; break; } } if (ownRow < 0 || aRows[ownRow - 1][6] !== "PENDING") { return ContentService.createTextOutput(JSON.stringify({ status: "cancelled_or_superseded" })) .setMimeType(ContentService.MimeType.JSON); } // Verify 2: Check Floorwalk_Logs directly — if record was edited and now has 0 FAIL/Remarked rooms, abort email!
